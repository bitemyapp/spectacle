include(GNUInstallDirs)
add_library(spectacle-fast-core STATIC FastCapture.cpp FastCapture.h)
target_link_libraries(spectacle-fast-core PUBLIC Qt6::Gui Qt6::DBus KF6::GuiAddons)
target_include_directories(spectacle-fast-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(spectacle-fast Main.cpp)
target_link_libraries(spectacle-fast PRIVATE spectacle-fast-core KF6::GlobalAccel)
install(TARGETS spectacle-fast RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../icons/sc-apps-spectacle.svg
    DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME spectacle-fast.svg)
configure_file(org.kde.spectacle.fast.desktop.in org.kde.spectacle.fast.desktop @ONLY)
configure_file(spectacle-fast.service.in spectacle-fast.service @ONLY)
configure_file(org.kde.Spectacle.Fast.service.in org.kde.Spectacle.Fast.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.kde.spectacle.fast.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.kde.Spectacle.Fast.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/spectacle-fast.service DESTINATION ${CMAKE_INSTALL_DATADIR}/systemd/user)
install(PROGRAMS setup-shortcut.sh DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME spectacle-fast-setup)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(spectacle-fast-tests Tests.cpp)
    target_link_libraries(spectacle-fast-tests PRIVATE spectacle-fast-core Qt6::Test)
    add_test(NAME spectacle-fast-geometry COMMAND spectacle-fast-tests)
    set_tests_properties(spectacle-fast-geometry PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
option(SPECTACLE_FAST_LIVE_TESTING "Build the private interactive test bridge (never installed)" OFF)
if(SPECTACLE_FAST_LIVE_TESTING)
    add_executable(spectacle-fast-probe LiveProbe.cpp)
    target_link_libraries(spectacle-fast-probe PRIVATE spectacle-fast-core)
endif()
